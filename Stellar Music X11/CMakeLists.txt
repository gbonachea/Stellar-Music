cmake_minimum_required(VERSION 3.16)
project(StellarMusic VERSION 2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Sql Concurrent DBus)
find_package(PkgConfig REQUIRED)
pkg_check_modules(AUDIO REQUIRED IMPORTED_TARGET gstreamer-1.0 gstreamer-fft-1.0 taglib)
add_library(hero_core OBJECT audioplayer.cpp playlistwindow.cpp configwindow.cpp settings.cpp playback.cpp display.cpp library.cpp mpris.cpp language.cpp transportbutton.h assets.qrc)
target_link_libraries(hero_core PUBLIC Qt6::Widgets Qt6::Sql Qt6::Concurrent Qt6::DBus PkgConfig::AUDIO)
target_include_directories(hero_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(StellarMusic main.cpp)
option(STELLAR_X11 "Use the Qt X11 backend by default" OFF)
if(STELLAR_X11)
  target_compile_definitions(StellarMusic PRIVATE STELLAR_X11)
endif()
target_link_libraries(StellarMusic PRIVATE hero_core)
option(HERO_TESTS "Build integration checks" OFF)
if(HERO_TESTS)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  add_executable(hero-check tests/check.cpp)
  target_link_libraries(hero-check PRIVATE hero_core Qt6::Test)
endif()
install(TARGETS StellarMusic DESTINATION bin)
install(FILES stellarmusic.desktop DESTINATION share/applications)
install(FILES icons/stellarmusic.png DESTINATION share/pixmaps RENAME stellarmusic.png)
